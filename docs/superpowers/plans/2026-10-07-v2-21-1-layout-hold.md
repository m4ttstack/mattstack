## Review Focus

1. **A held pull must not tell a member to reset.** The tag's `teamSyncRow` turns any `lastPullSkipped` on a pull-only clone into `needs-you` "cannot fast-forward (...); reset it to origin". A held pull sets `lastPullSkipped`, so without a guard the member is told to reset onto the layout their app cannot read. Task 3 excludes a held entry from that branch and from the "Last pull skipped" note, and tests it.
2. **A tip with no marker, an unparseable marker, or `role: "team"` passes.** The gate guards one thing; a clone with no readable marker is a `team.sync` problem. Task 2 tests a missing marker (git show reports "does not exist") and a `role: "team"` tip.
3. **The hold clears.** A later pull that passes the gate (origin moved back, or up to date) clears `layoutHold`, and the log-once latch resets so a second hold logs again. Task 2 tests both.
4. **A gate that throws (git show fails for a reason other than a missing path) never fast-forwards that tick.** It skips the tick with a plain detail, warns once per distinct message, and keeps any existing hold. Task 2 tests it.
5. **An older daemon's status entries have no `layoutHold`.** The row reads the field as optional and treats absent as no hold. Task 3 tests an entry without the field.

```ts
import { join } from "path";
import { stripJsonc } from "../jsonc.ts";
import { validateSlug } from "../secrets/store.ts";
import type { Probes } from "../setup/probes.ts";

/** The highest org layout this rt reads. Moves only with a breaking change to the org repo's shape, never with a release. */
export const ORG_LAYOUT = 1;

export const ORG_MARKER_REL = join("mattstack", "mattstack.jsonc");

export type MarkerState = { kind: "none" } | { kind: "invalid"; why: string } | { kind: "org"; org: string; layout: number };

/** A marker without a `layout` field reads 2 for the org layout and 1 for the one-team layout, so no repo has to write the field to be recognised. */
export function parseMarker(raw: string | null): MarkerState {
  if (raw === null) return { kind: "none" };
  let marker: unknown;
  try {
    marker = JSON.parse(stripJsonc(raw));
  } catch {
    return { kind: "invalid", why: "it is not valid JSON" };
  }
  if (!marker || typeof marker !== "object" || Array.isArray(marker)) return { kind: "invalid", why: "it is not a JSON object" };
  const { role, org, layout } = marker as Record<string, unknown>;
  if (role !== "org" && role !== "team") return { kind: "none" };
  if (typeof org !== "string") return { kind: "invalid", why: "it names no org" };
  try {
    validateSlug(org);
  } catch {
    return { kind: "invalid", why: `${JSON.stringify(org)} is not a valid org name` };
  }
  if (layout !== undefined && (typeof layout !== "number" || !Number.isInteger(layout) || layout < 1)) {
    return { kind: "invalid", why: "its layout is not a positive whole number" };
  }
  return { kind: "org", org, layout: typeof layout === "number" ? layout : role === "org" ? 2 : 1 };
}

export function markerState(p: Pick<Probes, "readFile">, dir: string): MarkerState {
  return parseMarker(p.readFile(join(dir, ORG_MARKER_REL)));
}

export function updateSentence(layout: number): string {
  return `Your org uses layout ${layout} and this app reads up to ${ORG_LAYOUT}. Update the app.`;
}
```

Gate semantics in `doPull`, after the `conflicted` checks and before the fast-forward / rebase:
- `behind === 0`: clear the hold (`layoutHold = null; loggedHold = null`) and return `up-to-date`. The gate is not called.
- `behind > 0` and `spec.pull.gate` set: call `gate("refs/remotes/origin/<branch>")`.
  - Non-null: `layoutHold = { layout, reads: ORG_LAYOUT }`; log at `info` only when `loggedHold !== layout` (then set `loggedHold = layout`); return `{ outcome: "skipped", detail: updateSentence(layout), hold: layoutHold }`.

  - Throws: warn once per distinct message (`loggedGateError`), keep `layoutHold` as it is, return `{ outcome: "skipped", detail: "could not read the org layout at origin: <message>" , ...(layoutHold ? { hold: layoutHold } : {}) }`. Never fast-forward on a throw.
- No gate (home snapshot): unchanged behaviour.

export async function layoutGate(exec: Probes["exec"], repoDir: string, ref: string): Promise<{ layout: number } | null> {
  const shown = await exec(["git", "-C", repoDir, "show", `${ref}:${ORG_MARKER_REL}`], { timeoutMs: GIT_TIMEOUT_MS, env: { LC_ALL: "C" } });

  }

}

export const ORG_LAYOUT_ROW_ID = "org.layout";
const UPDATE_APP_ACTION: Action = { type: "steps", label: "Show steps…", steps: ["Update mattstack from its menu bar icon, then reopen Setup status"] };

/** The team clone's layout against what this rt reads. A daemon holding the pull, or a clone already past this rt, needs the app update. */
export async function orgLayoutRow(
  p: Pick<Probes, "home" | "readDir" | "readFile" | "exists">,
  readStatus: () => Promise<TeamSnapshotEntry[] | null>,
): Promise<Row | null> {
  const teamsDir = join(p.home, ".mattstack", "teams");
  let folder: string | null = null;
  let layout = 0;
  for (const name of [...p.readDir(teamsDir)].sort()) {
    const dir = join(teamsDir, name);
    if (!p.exists(join(dir, ".git"))) continue;
    const marker = markerState(p, dir);
    if (marker.kind !== "org") continue;
    folder = name;
    layout = marker.layout;
    break;
  }
  if (folder === null) return null;
  const base = {
    id: ORG_LAYOUT_ROW_ID,
    kind: "tool" as const,
    title: "Org layout",
    why: "The org repo says what shape its files are in, and rt reads the shapes it knows. A newer shape waits for an app update.",
    required: false,
    recheck: "on-activate" as const,
  };
  const hold = (await readStatus())?.find((e) => e.slug === folder)?.layoutHold ?? null;
  if (hold) return row({ ...base, status: "needs-you", detail: updateSentence(hold.layout), action: UPDATE_APP_ACTION });
  if (layout > ORG_LAYOUT) return row({ ...base, status: "needs-you", detail: updateSentence(layout), action: UPDATE_APP_ACTION });
  return row({ ...base, status: "ready", detail: `${folder} on layout ${ORG_LAYOUT}` });
}
