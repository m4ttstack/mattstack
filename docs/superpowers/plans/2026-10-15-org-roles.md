# Org roles (plan 1 of 4) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the org owner role above admins, and let a team owner change who is on their own team, enforced by one org store guard used everywhere the org store is written, committed or published.

**Architecture:** `OrgRoles` gains `owners`; `roleOf` returns a new `org-owner` kind (admins count as owners while the list is empty). One pure guard, `orgStoreChangeRefusal(role, before, after)`, decides whether a change to the org store is this role's to make. `setSetting`, the daemon's team snapshot scope and `publishTeam` all call it, so the board's members screen and `rt team members` work for owners with no new daemon verb.

**Tech Stack:** Bun, TypeScript, bun:test (rt and rt-client), vitest (console, settings-kit).

**Spec:** `docs/superpowers/specs/2026-10-15-org-data-branch-and-roles-design.md` (sections "Roles" and "Write rules"). Plans 2 to 4 (data branch, forge access, admin-only secrets) follow; nothing here depends on them.

## Global Constraints

- No em dashes or en dashes anywhere (code comments, copy, docs).
- This repo is public: placeholders only (`acme`, `widgets`, `gadgets`, `dev1`), never a real company, person or repo.
- Usernames compare without case, always through `sameUser` (`packages/rt-client/src/settings/active-team.ts`).
- Every person-facing sentence follows the output layer rules in AGENTS.md: plain copy, commands in `next`, refusals by policy are `refused`, never `failed`.
- Every caller that asks "is this an admin" uses the new `isOrgAdmin(role)` helper; no new inline `kind === "admin"` test.
- An org whose `mattstack.org` has no `owners` (absent or empty) treats its admins as org owners.
- Run bun tests from the repo root (`bun test <path>`); console and settings-kit tests through `bun run console:test` / their package scripts.
- After touching `packages/rt-client/src`, run `bun run build` in `packages/rt-client` (dist goes stale silently).
- Commit after each task; only the files the task touched.

## Review Focus

1. An owner's edit that changes another team's membership in the same write (e.g. moves a member from `gadgets` to `widgets` when they own only `widgets`) must be refused whole, not half-applied.
2. An owner's edit that renames a member, changes an `agePublicKey`, or adds/removes a roster entry must be refused: only `teams` arrays may change.
3. A plain admin who edits `owners` through any path (setSetting, a hand edit the daemon would auto-commit, a publish) is refused; an org owner is not.
4. Usernames that differ only in case (`Dev2` vs `dev2`) match the same roster entry and the same owner.
5. A hand-edited org store holding an unrelated change plus a valid roster change is refused for an owner (the guard compares the whole store, not just the roster).

Each item has a test in the task that owns it (Task 2 for 1, 2, 4, 5; Tasks 2, 4, 5, 6 for 3).

```ts
/**
 * Whether a change to the org store is this role's to make. The org store is
 * one file, so path ownership cannot express "a team owner changes only their
 * own team's membership"; this compares the two versions instead.
 */

import { isDeepStrictEqual } from "util";
import { rolesFrom, rosterFrom, sameUser, type RosterEntry } from "./active-team.ts";
import { isOrgAdmin, type OrgRole } from "./org-roles.ts";
import type { StoreFile } from "./stores.ts";

const ROSTER = "mattstack.roster";

/** The store with the roster key removed, in both the flat and the nested spelling. */
function withoutKeys(store: StoreFile, keys: string[]): unknown {
  const global = structuredClone(store.global);
  for (const key of keys) {
    delete global[key];
    const [head, tail] = [key.slice(0, key.indexOf(".")), key.slice(key.indexOf(".") + 1)];
    const nested = global[head];
    if (nested !== null && typeof nested === "object" && !Array.isArray(nested)) delete (nested as Record<string, unknown>)[tail];
  }
  return { global, repos: store.repos };
}

function teamsOf(entry: RosterEntry): Set<string> {
  return new Set(Array.isArray(entry.teams) ? entry.teams.filter((t): t is string => typeof t === "string") : []);
}

function rosterRefusal(ownTeams: string[], before: RosterEntry[], after: RosterEntry[]): string | null {
  if (before.length !== after.length) return "A team owner can't add people to the org or remove them from it. An org admin does that.";
  for (const was of before) {
    const now = after.find((e) => sameUser(e.username, was.username));
    if (!now) return "A team owner can't add people to the org or remove them from it. An org admin does that.";
    const { teams: _a, ...wasRest } = was;
    const { teams: _b, ...nowRest } = now;
    if (!isDeepStrictEqual({ ...wasRest, username: wasRest.username.toLowerCase() }, { ...nowRest, username: nowRest.username.toLowerCase() })) {
      return `A team owner can only change which teams someone is on, not ${was.username}'s other details.`;
    }
    const a = teamsOf(was);
    const b = teamsOf(now);
    const changed = [...a].filter((t) => !b.has(t)).concat([...b].filter((t) => !a.has(t)));
    const foreign = changed.filter((t) => !ownTeams.includes(t));
    if (foreign.length > 0) return `You don't own ${foreign.join(" or ")}, so you can't change who is on it.`;
  }
  return null;
}

export function orgStoreChangeRefusal(role: OrgRole, before: StoreFile, after: StoreFile): string | null {
  if (role.kind === "org-owner") return null;
  if (role.kind === "admin") {
    const changed = !isDeepStrictEqual([...rolesFrom(before).owners].sort(), [...rolesFrom(after).owners].sort());
    return changed ? "Only an org owner can change who the org owners are." : null;
  }
  if (!isDeepStrictEqual(withoutKeys(before, [ROSTER]), withoutKeys(after, [ROSTER]))) {
    return role.kind === "owner" ? "A team owner can only change which teams people are on in the org's settings." : "The org's shared settings belong to its admins.";
  }
  if (role.kind !== "owner") return isDeepStrictEqual(rosterFrom(before), rosterFrom(after)) ? null : "The org's shared settings belong to its admins.";
  return rosterRefusal(role.teams, rosterFrom(before), rosterFrom(after));
}
```

```ts
import { readFileSync } from "fs";
import { join } from "path";
import { orgStoreChangeRefusal } from "../../packages/rt-client/src/settings/org-store-guard.ts";
import type { OrgRole } from "../../packages/rt-client/src/settings/org-roles.ts";
import { parseStoreText } from "../settings/stores.ts";
import { childEnv } from "../subprocess.ts";

export const ORG_STORE_REL = "mattstack/org/settings.org.jsonc";

type Run = (argv: string[]) => { code: number; stdout: string };

const realRun: Run = (argv) => {
  const r = Bun.spawnSync(argv, { env: childEnv(), stdout: "pipe", stderr: "ignore" });
  return { code: r.exitCode ?? 1, stdout: r.stdout.toString() };
};

/** Read synchronously because `readAuthorization` is synchronous; the file is small. */
export function orgStoreInScope(repoDir: string, role: OrgRole, run: Run = realRun): boolean {
  let working: string;
  try {
    working = readFileSync(join(repoDir, ORG_STORE_REL), "utf8");
  } catch {
    return false;
  }
  const head = run(["git", "-C", repoDir, "show", `HEAD:${ORG_STORE_REL}`]);
  const headText = head.code === 0 ? head.stdout : "{}";
  if (headText === working) return true;
  return orgStoreChangeRefusal(role, parseStoreText(ORG_STORE_REL, headText), parseStoreText(ORG_STORE_REL, working)) === null;
}
```

In `teamSnapshotSpec`'s `readAuthorization` (members and unknown roles never get the org store in scope):

```ts
readAuthorization: () => {
  const role = roleFor(opts.probes, slug);
  const roots = ownedRoots(role);
  const shares = readTeamLocal(opts.probes, slug).pendingPackShares ?? [];
  const mayTouchOrgStore = role.kind === "owner" || isOrgAdmin(role);
  const orgStore = mayTouchOrgStore && orgStoreInScope(repoDir, role);
  const inRoots = (path: string) => roots.some((root) => path === root || path.startsWith(`${root}/`));
  return {
    scope: (path) => teamScope(path) && (path === ORG_STORE_REL ? orgStore : inRoots(path)),
    pullOnly: roots.length === 0,
    held: (path, dirty) => path === MARKETPLACE_PATH && marketplaceOwed(shares, repoDir, dirty),
  };
},
```

```ts
export type MarkerState = { kind: "none" } | { kind: "invalid"; why: string } | { kind: "org"; org: string; layout: number; dataBranch: string | null };
```

```ts
import { markerState } from "./org-marker.ts";

/** The branch that holds the org's data: the marker's `dataBranch` (layout 4 on), else the branch checked out. */
export async function dataBranchOf(p: Pick<Probes, "exec" | "readFile">, dir: string): Promise<string> {
  const state = markerState(p, dir);
  if (state.kind === "org" && state.dataBranch !== null) return state.dataBranch;
  return orgBranch(p, dir);
}
```
